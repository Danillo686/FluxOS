SELECT
    policyname,
    cmd,
    roles
FROM pg_policies
WHERE tablename = 'users';

CREATE POLICY "Permitir cadastro de clientes" // Quem estiver sujeito ao RLS pode inserir uma linha em users, desde que ela tenha role = 'cliente'.
ON public.users
FOR INSERT
WITH CHECK (
    role = 'cliente'
);


CREATE POLICY "Usuário pode consultar seu próprio perfil"
ON public.users
FOR SELECT
USING (
    id = auth.uid()
);
